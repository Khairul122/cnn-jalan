' 1. INPUT DATA LOKASI
' ============================================================
@startuml SD_01_InputDataLokasi
title Sequence Diagram — Input Data Lokasi

actor    Admin
boundary Browser
control  LokasiController
database Database

Admin -> Browser : Isi form lokasi + upload foto
activate Browser

Browser -> LokasiController : POST /lokasi/create
activate LokasiController

LokasiController -> LokasiController : Validasi input

alt Data valid
    LokasiController -> Database : INSERT lokasi_kerusakan\n+ dokumentasi_foto
    activate Database
    Database --> LokasiController : OK
    deactivate Database

    LokasiController --> Browser : Redirect + flash sukses
    deactivate LokasiController

    deactivate Browser

else Data tidak valid
    LokasiController --> Browser : Render form + pesan error
    deactivate LokasiController
    Browser --> Admin : Tampil pesan error
    deactivate Browser
end

' ============================================================

    PreprocessingController --> Browser : Redirect + flash reset berhasil

' ============================================================
' 5. TRAINING CNN
' ============================================================
@startuml SD_05_TrainingCNN
title Sequence Diagram — Training CNN

actor    Admin
boundary Browser
control  ArsitekturController
entity   CNNService
database Database

Admin -> Browser : Isi hyperparameter → Klik Mulai Training
activate Browser

Browser -> ArsitekturController : POST /arsitektur/create
activate ArsitekturController

ArsitekturController -> Database : INSERT arsitektur_config\n(status = antri)
activate Database
Database --> ArsitekturController : arsitektur_id
deactivate Database

ArsitekturController -> CNNService : train(arsitektur) [background thread]
activate CNNService

ArsitekturController --> Browser : Redirect ke halaman monitor progress
deactivate ArsitekturController
Browser --> Admin : Tampil halaman monitor training
deactivate Browser

CNNService -> CNNService : Build model + load dataset\nHitung class_weight
CNNService -> CNNService : Phase 1 — Head Training\nPhase 2 — Fine-tuning

CNNService -> Database : INSERT hasil_training (per epoch)
activate Database
Database --> CNNService : OK
deactivate Database

CNNService -> CNNService : Auto Evaluate (confusion matrix + F1)

CNNService -> Database : INSERT hasil_evaluasi\nINSERT prediksi_model\nUPDATE status = selesai
activate Database
Database --> CNNService : OK
deactivate Database

deactivate CNNService

Admin -> Browser : Polling GET /arsitektur/<id>/progress
activate Browser
Browser -> ArsitekturController : GET /progress
activate ArsitekturController

alt Training sedang berjalan
    ArsitekturController --> Browser : JSON (epoch, loss, val_accuracy)
    deactivate ArsitekturController
    Browser --> Admin : Update grafik live
    deactivate Browser
else Training selesai
    ArsitekturController --> Browser : JSON (status = selesai, akurasi)
    deactivate ArsitekturController
    Browser --> Admin : Tampil notifikasi selesai
    deactivate Browser
end

@enduml

    ArsitekturController --> Browser : Render halaman evaluasi
    deactivate ArsitekturController

    deactivate Browser

    Admin -> Browser : Analisis hasil
    activate Browser

    alt Akurasi memenuhi target
        Browser --> Admin : Lanjut ke Peta GIS
        deactivate Browser
    else Akurasi belum memenuhi target
        Browser --> Admin : Kembali atur hyperparameter\ndan training ulang
        deactivate Browser
    end

else Data belum tersedia
    ArsitekturController --> Browser : Flash "Jalankan training dahulu"
    deactivate ArsitekturController
    Browser --> Admin : Tampil pesan informasi
    deactivate Browser
end
